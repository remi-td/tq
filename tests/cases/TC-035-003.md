# TC-035-003: Project Config TOML Structure and Parsing

| **Test Type** | Unit |
| **Created** | 2026-02-13 |

## Purpose

Verify that project configuration files (`.tq.toml`) use the same TOML structure as user configuration and are parsed correctly, with comprehensive error handling.

## Acceptance Criteria Coverage

- **AC-3**: Support same TOML structure as user config (profiles, preferences) - TOML compatibility
- **AC-7**: Comprehensive error handling (invalid TOML, permission errors) - Error handling

## Scope

This test validates:
- `.tq.toml` has same structure as `~/.tq/config.toml`
- Profiles section parses correctly
- Defaults section parses correctly
- Invalid TOML produces clear error messages
- Missing required fields are handled
- Extra unknown fields are ignored (or warned)
- File read errors are handled gracefully

## Prerequisites

- Rust test framework available
- tempfile crate available for test file creation
- TOML parsing implementation exists

## Test Procedure

### Test Implementation:

```rust
#[cfg(test)]
mod tests {
    use super::*;
    use std::fs;
    use tempfile::TempDir;

    #[test]
    fn test_parse_project_config_valid_profiles() {
        let toml_content = r#"
            [profiles.dev]
            host = "dev.company.com"
            port = 1025
            database = "development"

            [profiles.staging]
            host = "staging.company.com"
            database = "staging"
        "#;

        let config: Config = toml::from_str(toml_content).unwrap();

        assert_eq!(config.profiles.len(), 2);
        assert!(config.profiles.contains_key("dev"));
        assert!(config.profiles.contains_key("staging"));

        let dev = config.profiles.get("dev").unwrap();
        assert_eq!(dev.host.as_ref().unwrap(), "dev.company.com");
        assert_eq!(dev.port.unwrap(), 1025);
        assert_eq!(dev.database.as_ref().unwrap(), "development");

        let staging = config.profiles.get("staging").unwrap();
        assert_eq!(staging.host.as_ref().unwrap(), "staging.company.com");
        assert_eq!(staging.database.as_ref().unwrap(), "staging");
    }

    #[test]
    fn test_parse_project_config_valid_defaults() {
        let toml_content = r#"
            [defaults]
            format = "table"
            timing = true
            paging = false
        "#;

        let config: Config = toml::from_str(toml_content).unwrap();

        assert!(config.defaults.is_some());
        let defaults = config.defaults.unwrap();
        assert_eq!(defaults.format.as_ref().unwrap(), "table");
        assert_eq!(defaults.timing.unwrap(), true);
        assert_eq!(defaults.paging.unwrap(), false);
    }

    #[test]
    fn test_parse_project_config_both_sections() {
        let toml_content = r#"
            [defaults]
            format = "json"
            timing = true

            [profiles.dev]
            host = "dev.company.com"
            database = "development"

            [profiles.prod]
            host = "prod.company.com"
            database = "production"
            logmech = "LDAP"
        "#;

        let config: Config = toml::from_str(toml_content).unwrap();

        // Defaults present
        assert!(config.defaults.is_some());
        assert_eq!(
            config.defaults.as_ref().unwrap().format.as_ref().unwrap(),
            "json"
        );

        // Profiles present
        assert_eq!(config.profiles.len(), 2);
        assert!(config.profiles.contains_key("dev"));
        assert!(config.profiles.contains_key("prod"));
    }

    #[test]
    fn test_parse_project_config_empty() {
        let toml_content = r#""#;

        let config: Config = toml::from_str(toml_content).unwrap();

        // Empty config is valid
        assert!(config.defaults.is_none());
        assert!(config.profiles.is_empty());
    }

    #[test]
    fn test_parse_project_config_invalid_toml_syntax() {
        let toml_content = r#"
            [profiles.dev]
            host: "dev.company.com"  # Wrong syntax - should use '='
        "#;

        let result: Result<Config, _> = toml::from_str(toml_content);

        assert!(result.is_err(), "Invalid TOML should produce error");
        let error_msg = result.unwrap_err().to_string();
        assert!(
            error_msg.contains("expected") || error_msg.contains("="),
            "Error should mention syntax issue"
        );
    }

    #[test]
    fn test_parse_project_config_invalid_type() {
        let toml_content = r#"
            [defaults]
            timing = "yes"  # Should be boolean, not string
        "#;

        let result: Result<Config, _> = toml::from_str(toml_content);

        assert!(result.is_err(), "Type mismatch should produce error");
        let error_msg = result.unwrap_err().to_string();
        assert!(
            error_msg.contains("invalid type") || error_msg.contains("boolean"),
            "Error should mention type mismatch"
        );
    }

    #[test]
    fn test_parse_project_config_invalid_port() {
        let toml_content = r#"
            [profiles.dev]
            host = "dev.company.com"
            port = "not-a-number"  # Should be integer
        "#;

        let result: Result<Config, _> = toml::from_str(toml_content);

        assert!(result.is_err(), "Non-integer port should produce error");
    }

    #[test]
    fn test_load_project_config_from_file() {
        // Create temp file with valid config
        let temp_dir = TempDir::new().unwrap();
        let config_path = temp_dir.path().join(".tq.toml");

        let toml_content = r#"
            [defaults]
            format = "csv"

            [profiles.test]
            host = "test.example.com"
        "#;
        fs::write(&config_path, toml_content).unwrap();

        // Load config
        let config = load_config_from_file(&config_path).unwrap();

        assert!(config.defaults.is_some());
        assert_eq!(
            config.defaults.as_ref().unwrap().format.as_ref().unwrap(),
            "csv"
        );
        assert!(config.profiles.contains_key("test"));
    }

    #[test]
    fn test_load_project_config_file_not_found() {
        let temp_dir = TempDir::new().unwrap();
        let config_path = temp_dir.path().join("nonexistent.toml");

        let result = load_config_from_file(&config_path);

        assert!(result.is_err(), "Missing file should produce error");
        let error = result.unwrap_err();
        assert!(
            error.to_string().contains("No such file") || error.to_string().contains("not found"),
            "Error should indicate file not found"
        );
    }

    #[test]
    fn test_load_project_config_permission_denied() {
        // Create temp file with no read permissions
        let temp_dir = TempDir::new().unwrap();
        let config_path = temp_dir.path().join(".tq.toml");
        fs::write(&config_path, "[defaults]\nformat = \"json\"").unwrap();

        #[cfg(unix)]
        {
            use std::os::unix::fs::PermissionsExt;
            let mut perms = fs::metadata(&config_path).unwrap().permissions();
            perms.set_mode(0o000); // No permissions
            fs::set_permissions(&config_path, perms).unwrap();

            let result = load_config_from_file(&config_path);

            assert!(result.is_err(), "Permission denied should produce error");
            let error = result.unwrap_err();
            assert!(
                error.to_string().contains("permission") || error.to_string().contains("denied"),
                "Error should indicate permission denied"
            );

            // Clean up: restore permissions so temp dir can be deleted
            let mut perms = fs::metadata(&config_path).unwrap().permissions();
            perms.set_mode(0o644);
            fs::set_permissions(&config_path, perms).unwrap();
        }

        #[cfg(not(unix))]
        {
            // Skip test on non-Unix systems (Windows has different permission model)
            println!("Skipping permission test on non-Unix platform");
        }
    }

    #[test]
    fn test_parse_project_config_partial_profile() {
        // Profile with only host (minimal valid profile)
        let toml_content = r#"
            [profiles.minimal]
            host = "example.com"
        "#;

        let config: Config = toml::from_str(toml_content).unwrap();

        assert_eq!(config.profiles.len(), 1);
        let profile = config.profiles.get("minimal").unwrap();
        assert_eq!(profile.host.as_ref().unwrap(), "example.com");
        assert!(profile.port.is_none());
        assert!(profile.database.is_none());
        assert!(profile.user.is_none());
    }

    #[test]
    fn test_parse_project_config_complete_profile() {
        // Profile with all fields
        let toml_content = r#"
            [profiles.complete]
            host = "example.com"
            port = 1025
            database = "mydb"
            user = "alice"
            logmech = "LDAP"
            password_file = "~/.tq/passwords/complete"
            timeout = "60s"
        "#;

        let config: Config = toml::from_str(toml_content).unwrap();

        let profile = config.profiles.get("complete").unwrap();
        assert_eq!(profile.host.as_ref().unwrap(), "example.com");
        assert_eq!(profile.port.unwrap(), 1025);
        assert_eq!(profile.database.as_ref().unwrap(), "mydb");
        assert_eq!(profile.user.as_ref().unwrap(), "alice");
        assert_eq!(profile.logmech.as_ref().unwrap(), "LDAP");
        assert_eq!(
            profile.password_file.as_ref().unwrap(),
            "~/.tq/passwords/complete"
        );
        assert_eq!(profile.timeout.as_ref().unwrap(), "60s");
    }
}
```
