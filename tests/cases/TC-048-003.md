### Prerequisites

### Test Implementation

Add to `#[cfg(test)] mod tests` in `src/commands/inspect.rs`:

```rust
// TC-048-003-A1: summarize_error on pure ASCII — no change under 80 chars
#[test]
fn summarize_error_ascii_short() {
    let short_msg = "Connection refused to host 192.168.1.1:1025";
    // The function truncates at 80 chars; this is well under
    // We test the logic: if msg.len() <= 80, return as-is
    let result = if short_msg.len() > 80 {
        format!("{}...", &short_msg[..77])
    } else {
        short_msg.to_string()
    };
    assert_eq!(result, short_msg, "Short ASCII messages must not be modified");
}

// TC-048-003-A2: summarize_error on pure ASCII — truncated at exactly 80 chars
#[test]
fn summarize_error_ascii_truncation() {
    // Build a message exactly 100 ASCII characters long
    let long_msg = "A".repeat(100);
    // After fix: truncation at char boundary (safe for ASCII since 1 byte = 1 char)
    let result = if long_msg.len() > 80 {
        // UTF-8 safe: find the byte position of the 77th character
        let truncate_at = long_msg
            .char_indices()
            .nth(77)
            .map(|(i, _)| i)
            .unwrap_or(77);
        format!("{}...", &long_msg[..truncate_at])
    } else {
        long_msg.clone()
    };
    assert!(result.ends_with("..."), "Truncated message must end with '...'");
    assert_eq!(result.len(), 80, "Truncated message must be exactly 80 chars");
    assert!(std::str::from_utf8(result.as_bytes()).is_ok(), "Result must be valid UTF-8");
}

// TC-048-003-A3: summarize_error with 2-byte UTF-8 characters (é = 0xC3 0xA9)
// Without fix: &msg[..77] could split 'é' and produce invalid UTF-8 or panic
#[test]
fn summarize_error_utf8_2byte_chars() {
    // 'é' is 2 bytes in UTF-8; build a string of 50 'é' characters = 100 bytes
    let long_msg = "é".repeat(50);
    assert_eq!(long_msg.len(), 100, "50 × é = 100 bytes");

    // UTF-8 safe truncation
    let result = if long_msg.len() > 80 {
        let truncate_at = long_msg
            .char_indices()
            .nth(77)
            .map(|(i, _)| i)
            .unwrap_or(long_msg.len());
        format!("{}...", &long_msg[..truncate_at])
    } else {
        long_msg.clone()
    };

    // The result must be valid UTF-8 (no panic, no invalid bytes)
    assert!(
        std::str::from_utf8(result.as_bytes()).is_ok(),
        "Result must be valid UTF-8 after truncation"
    );
    assert!(result.ends_with("..."), "Truncated message must end with '...'");
    // Each 'é' is 2 bytes; 77 chars × 2 bytes + 3 "..." = 157 bytes — well under panic range
    let char_count = result.chars().count() - 3; // subtract "..."
    assert!(
        char_count <= 77,
        "Must not exceed 77 characters before '...': got {}",
        char_count
    );
}

// TC-048-003-A4: summarize_error with 3-byte UTF-8 characters (中 = 0xE4 0xB8 0xAD)
#[test]
fn summarize_error_utf8_3byte_chars() {
    // '中' is 3 bytes; 30 × '中' = 90 bytes (> 80), 27 × '中' = 81 bytes (> 80)
    let long_msg = "中".repeat(30);
    assert_eq!(long_msg.len(), 90, "30 × '中' = 90 bytes");

    let result = if long_msg.len() > 80 {
        let truncate_at = long_msg
            .char_indices()
            .nth(77)
            .map(|(i, _)| i)
            .unwrap_or(long_msg.len());
        format!("{}...", &long_msg[..truncate_at])
    } else {
        long_msg.clone()
    };

    assert!(
        std::str::from_utf8(result.as_bytes()).is_ok(),
        "Result must be valid UTF-8 after 3-byte char truncation"
    );
    assert!(result.ends_with("..."), "Must end with '...'");
}

// TC-048-003-A5: summarize_error with 4-byte UTF-8 characters (emoji: 😀 = 4 bytes)
#[test]
fn summarize_error_utf8_4byte_emoji() {
    // '😀' is 4 bytes; 21 × '😀' = 84 bytes (> 80)
    let long_msg = "😀".repeat(21);
    assert_eq!(long_msg.len(), 84, "21 × emoji = 84 bytes");

    let result = if long_msg.len() > 80 {
        let truncate_at = long_msg
            .char_indices()
            .nth(77)
            .map(|(i, _)| i)
            .unwrap_or(long_msg.len());
        format!("{}...", &long_msg[..truncate_at])
    } else {
        long_msg.clone()
    };

    assert!(
        std::str::from_utf8(result.as_bytes()).is_ok(),
        "Result must be valid UTF-8 after 4-byte char truncation"
    );
    // With 21 emoji (84 bytes), 77 chars < 21, so the truncation is at the full string
    // (21 × 4 = 84 bytes > 80, but 21 chars ≤ 77 chars, so we may not truncate by char count)
    // The important thing: no panic, valid UTF-8
}

// TC-048-003-A6: summarize_error exact boundary — message of exactly 80 bytes but < 80 chars
#[test]
fn summarize_error_80_bytes_with_multibyte() {
    // Mix ASCII (1 byte) and 'é' (2 bytes) to get exactly 80 bytes total
    // 40 ASCII + 20 'é' = 40 + 40 = 80 bytes = 60 characters
    let mut long_msg = String::new();
    for _ in 0..40 { long_msg.push('A'); }
    for _ in 0..20 { long_msg.push('é'); }
    assert_eq!(long_msg.len(), 80, "Must be exactly 80 bytes");
    assert_eq!(long_msg.chars().count(), 60, "Must be 60 characters");

    // At exactly 80 bytes: msg.len() == 80, NOT > 80, so no truncation
    let result = if long_msg.len() > 80 {
        let truncate_at = long_msg
            .char_indices()
            .nth(77)
            .map(|(i, _)| i)
            .unwrap_or(long_msg.len());
        format!("{}...", &long_msg[..truncate_at])
    } else {
        long_msg.clone()
    };

    assert_eq!(result, long_msg, "Messages of exactly 80 bytes must not be truncated");
    assert!(std::str::from_utf8(result.as_bytes()).is_ok(), "Must be valid UTF-8");
}
```

```rust
// TC-048-003-D1: list unknown subcommand output starts with "Error:"
#[test]
fn list_unknown_subcommand_has_error_prefix() {
    use std::io::Write;
    // Simulate the unknown subcommand output (post-fix)
    let mut buf: Vec<u8> = Vec::new();
    let subcommand = "bogus";
    writeln!(buf, "Error: Unknown list subcommand: {}", subcommand).unwrap();
    writeln!(buf, "Available: databases, tables, views").unwrap();

    let output = String::from_utf8(buf).unwrap();
    assert!(
        output.starts_with("Error:"),
        "Unknown subcommand output must start with 'Error:': {}",
        output
    );
    assert!(
        output.contains("bogus"),
        "Error message must name the unknown subcommand: {}",
        output
    );
    assert!(
        output.contains("Available:"),
        "Error message must list available subcommands: {}",
        output
    );
}
```
