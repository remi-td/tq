// TC-047-001-A / TC-048-005-A1
// Multi-row DDL concatenation — view
// Simulates SHOW VIEW returning 3 rows (Teradata splits long DDL across rows)
#[test]
fn ddl_multirow_all_rows_concatenated() {
    let rows = vec![
        "CREATE VIEW \"DBC\".\"Tables\" AS".to_string(),
        " SELECT TableName, DatabaseName".to_string(),
        " FROM DBC.TVM WHERE TVMNameI = 'T'".to_string(),
    ];
    let definition = rows.join("");
    assert!(definition.contains("CREATE VIEW"), "First row must be present");
    assert!(definition.contains("FROM DBC.TVM"), "Last row must be present");
    assert!(
        !definition.is_empty(),
        "Multi-row result must not produce empty definition"
    );
}

// TC-047-001-B / TC-048-005-A2
// NULL row filtering — rows with "[NULL]" are excluded
#[test]
fn ddl_null_rows_filtered_out() {
    let raw_values = vec!["CREATE VIEW db.v AS SELECT 1".to_string(), "[NULL]".to_string()];
    let definition: String = raw_values
        .into_iter()
        .filter(|s| s != "[NULL]")
        .collect::<Vec<_>>()
        .join("");
    assert_eq!(definition, "CREATE VIEW db.v AS SELECT 1");
    assert!(!definition.contains("[NULL]"), "[NULL] rows must not appear in definition");
}

// TC-047-001-C / TC-048-005-A3
// Empty result — SHOW VIEW returns 0 rows → empty definition (no panic)
#[test]
fn ddl_empty_result_returns_empty_string() {
    let rows: Vec<String> = vec![];
    let definition: String = rows.join("");
    assert_eq!(definition, "", "Empty result must produce empty definition");
    // No panic is the implicit assertion here
}

// TC-047-001-D / TC-048-005-A4
// View SQL uses SHOW VIEW (not SHOW TABLE)
#[test]
fn ddl_view_sql_uses_show_view() {
    let db = "DBC";
    let obj = "Tables";
    let kind = "V";
    let sql = match kind {
        "V" => format!("SHOW VIEW \"{}\".\"{}\"", db, obj),
        "M" => format!("SHOW MACRO \"{}\".\"{}\"", db, obj),
        _ => String::new(),
    };
    assert!(sql.starts_with("SHOW VIEW"), "View DDL query must use SHOW VIEW");
    assert!(sql.contains("\"DBC\""), "Database part must be quoted");
    assert!(sql.contains("\"Tables\""), "Object part must be quoted");
}

// TC-047-001-E / TC-048-005-A5
// Macro SQL uses SHOW MACRO (not SHOW VIEW)
#[test]
fn ddl_macro_sql_uses_show_macro() {
    let db = "mydb";
    let obj = "mymacro";
    let kind = "M";
    let sql = match kind {
        "V" => format!("SHOW VIEW \"{}\".\"{}\"", db, obj),
        "M" => format!("SHOW MACRO \"{}\".\"{}\"", db, obj),
        _ => String::new(),
    };
    assert!(sql.starts_with("SHOW MACRO"), "Macro DDL query must use SHOW MACRO");
    assert!(sql.contains("\"mydb\""), "Database part must be double-quoted");
    assert!(sql.contains("\"mymacro\""), "Object part must be double-quoted");
}

// TC-047-001-F / TC-048-005-A6
// Unknown kind returns empty definition (tables have no SHOW DDL)

    assert!(output.contains("System"), "System type label must appear: {}", output);
    assert!(output.contains("User"), "User type label must appear: {}", output);
    assert!(output.contains("2 database(s)"), "Count footer must appear: {}", output);
}

// TC-048-005-D2: list_databases JSON format — database key, correct types
#[test]
fn list_databases_json_writer_format() {
    use std::io::Write;
    let mut buf: Vec<u8> = Vec::new();

    let databases = vec![
        DatabaseEntry {
            name: "mydb".to_string(),
            owner: "admin".to_string(),
            db_kind: "User".to_string(),
        },
    ];

    write!(buf, "[").unwrap();
    for (i, db) in databases.iter().enumerate() {
        if i > 0 { write!(buf, ",").unwrap(); }
        write!(buf,
            "{{\"database\":\"{}\",\"owner\":\"{}\",\"type\":\"{}\"}}",
            db.name, db.owner, db.db_kind
        ).unwrap();
    }
    writeln!(buf, "]").unwrap();

    let json_str = String::from_utf8(buf).unwrap();
    assert!(json_str.contains("\"database\":\"mydb\""), "Must use 'database' key: {}", json_str);
    assert!(!json_str.contains("\"name\":\"mydb\""), "Must NOT use 'name' key: {}", json_str);
    assert!(json_str.contains("\"owner\":\"admin\""), "Owner must appear: {}", json_str);
    assert!(json_str.contains("\"type\":\"User\""), "Type must appear: {}", json_str);
}
