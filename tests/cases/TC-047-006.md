# TC-047-006: Enrich `tq show-indexes` Output

## Metadata

| Field | Value |
|-------|-------|
| **Test ID** | TC-047-006 |
| **Title** | `tq show-indexes` output uses Primary/Secondary sections, UPI/NUPI/USI/NUSI labels, and structured JSON |
| **Category** | Unit Test + Integration Test |
| **Priority** | High (P1) |
| **Feature** | Sprint 47 — Feature 6: Enrich `tq show-indexes` Output |
| **Test Type** | Unit (no DB) + Integration (DB required, `#[ignore]`) |
| **DB Required** | No for unit tests; Yes for integration tests |
| **Created** | 2026-03-23 |
| **Acceptance Criteria** | AC-1, AC-2, AC-3, AC-4 |

## Purpose

Verify that after the enrichment:
1. `tq show-indexes` output has a two-section layout: "Primary Index" section and "Secondary Indexes" section.
2. Index types are labeled as UPI, NUPI, USI, or NUSI — derived from combining `IndexType` (P/S) and `UniqueFlag` (Y/N).
3. JSON output uses a structured `{"object": ..., "primary_index": {...}, "secondary_indexes": [...]}` wrapper.
4. Unit tests exist for all formatting functions in `show_indexes.rs`.

Special cases:
- NoPI tables (no primary index rows): "No Primary Index (NoPI)" rendered in Primary section.
- Tables with no secondary indexes: "No secondary indexes defined." rendered.

## Acceptance Criteria Coverage

- **AC-1**: Separate Primary Index and Secondary Indexes sections — spec: cli-interface.md §show-indexes
- **AC-2**: Labels: UPI/NUPI/USI/NUSI — spec: cli-interface.md §show-indexes
- **AC-3**: JSON: `{primary_index, secondary_indexes[]}` — spec: cli-interface.md §show-indexes
- **AC-4**: Unit tests for show_indexes.rs formatting functions

#[cfg(test)]
mod tests {
    use super::*;

    // Helper to check label derivation: IndexType (P/S) + UniqueFlag (Y/N) → label
    fn derive_index_label(index_type: &str, unique_flag: &str) -> String {
        match (index_type, unique_flag) {
            ("P", "Y") | ("P", "U") => "UPI".to_string(),   // Unique Primary Index
            ("P", _)                 => "NUPI".to_string(),  // Non-Unique Primary Index
            ("S", "Y") | ("S", "U") => "USI".to_string(),   // Unique Secondary Index
            ("S", _)                 => "NUSI".to_string(),  // Non-Unique Secondary Index
            _                        => "Unknown".to_string(),
        }
    }

    fn derive_full_label(short_label: &str) -> String {
        match short_label {
            "UPI"  => "Unique Primary Index (UPI)".to_string(),
            "NUPI" => "Non-Unique Primary Index (NUPI)".to_string(),
            "USI"  => "Unique Secondary Index (USI)".to_string(),
            "NUSI" => "Non-Unique Secondary Index (NUSI)".to_string(),
            _      => short_label.to_string(),
        }
    }

    // TC-047-006-A: Output header contains object name
    #[test]
    fn show_indexes_header_contains_object_name() {
        let header = format!("Index structure for {}.{}:", "PRODUCTION", "employees");
        assert!(header.contains("Index structure for"), "Header must contain 'Index structure for'");
        assert!(header.contains("PRODUCTION"), "Header must contain database name");
        assert!(header.contains("employees"), "Header must contain object name");
        assert!(header.ends_with(':'), "Header must end with ':'");
    }

    // TC-047-006-B: Primary Index section header present
    #[test]
    fn show_indexes_primary_section_header_present() {
        let section = "  Primary Index";
        assert!(section.contains("Primary Index"), "Primary section header must be present");
    }

    // TC-047-006-C: UPI label — IndexType="P", UniqueFlag="Y" (AC-2)
    #[test]
    fn show_indexes_label_upi_for_primary_unique() {
        let label = derive_index_label("P", "Y");
        assert_eq!(label, "UPI", "Primary unique index must be labeled UPI");
        let full = derive_full_label("UPI");
        assert!(full.contains("Unique Primary Index"), "Full UPI label must be 'Unique Primary Index (UPI)'");
        assert!(full.contains("UPI"), "Full label must include UPI abbreviation");
    }

    // TC-047-006-D: NUPI label — IndexType="P", UniqueFlag="N" (AC-2)
    #[test]
    fn show_indexes_label_nupi_for_primary_nonunique() {
        let label = derive_index_label("P", "N");
        assert_eq!(label, "NUPI", "Primary non-unique index must be labeled NUPI");
        let full = derive_full_label("NUPI");
        assert!(full.contains("Non-Unique Primary Index"), "Full NUPI label must be 'Non-Unique Primary Index (NUPI)'");
        assert!(full.contains("NUPI"), "Full label must include NUPI abbreviation");
    }

    // TC-047-006-E: USI label — IndexType="S", UniqueFlag="Y" (AC-2)
    #[test]
    fn show_indexes_label_usi_for_secondary_unique() {
        let label = derive_index_label("S", "Y");
        assert_eq!(label, "USI", "Secondary unique index must be labeled USI");
        let full = derive_full_label("USI");
        assert!(full.contains("Unique Secondary Index"), "Full USI label must be 'Unique Secondary Index (USI)'");
        assert!(full.contains("USI"), "Full label must include USI abbreviation");
    }

    // TC-047-006-F: NUSI label — IndexType="S", UniqueFlag="N" (AC-2)
    #[test]
    fn show_indexes_label_nusi_for_secondary_nonunique() {
        let label = derive_index_label("S", "N");
        assert_eq!(label, "NUSI", "Secondary non-unique index must be labeled NUSI");
        let full = derive_full_label("NUSI");
        assert!(full.contains("Non-Unique Secondary Index"), "Full NUSI label must be 'Non-Unique Secondary Index (NUSI)'");
        assert!(full.contains("NUSI"), "Full label must include NUSI abbreviation");
    }

    // TC-047-006-G: NoPI — no primary index rows → "No Primary Index (NoPI)" (AC-1)
    #[test]
    fn show_indexes_no_primary_index_shows_nopi_message() {
        let primary_rows: Vec<String> = vec![];
        let has_primary = !primary_rows.is_empty();
        if !has_primary {
            let message = "No Primary Index (NoPI)";
            assert!(message.contains("NoPI"), "NoPI message must contain 'NoPI' text");
            assert!(message.contains("No Primary Index"), "NoPI message must describe absence of primary index");
        }
    }

    // TC-047-006-H: Secondary Indexes section header present (AC-1)
    #[test]
    fn show_indexes_secondary_section_header_present() {
        let section = "  Secondary Indexes";
        assert!(section.contains("Secondary Indexes"), "Secondary section header must be present");
    }

    // TC-047-006-I: Secondary indexes numbered (#1, #2, #3) (AC-1)
    #[test]
    fn show_indexes_secondary_indexes_numbered() {
        let mut output = Vec::<u8>::new();
        // Simulate numbering logic
        let secondary_indexes = vec!["NUSI: department_id", "USI: email"];
        for (i, idx) in secondary_indexes.iter().enumerate() {
            let line = format!("    #{}  {}", i + 1, idx);
            output.extend_from_slice(line.as_bytes());
            output.push(b'\n');
        }
        let rendered = String::from_utf8(output).unwrap();
        assert!(rendered.contains("#1"), "First secondary index must be labeled #1");
        assert!(rendered.contains("#2"), "Second secondary index must be labeled #2");
    }

    // TC-047-006-J: No secondary indexes → "No secondary indexes defined." message (AC-1)
    #[test]
    fn show_indexes_no_secondary_shows_message() {
        let secondary_indexes: Vec<String> = vec![];
        let message = if secondary_indexes.is_empty() {
            "No secondary indexes defined."
        } else {
            ""
        };
        assert_eq!(
            message,
            "No secondary indexes defined.",
            "Empty secondary indexes must show 'No secondary indexes defined.'"
        );
    }

    // TC-047-006-K: JSON has top-level "object" key (AC-3)
    #[test]
    fn show_indexes_json_has_object_key() {
        let json = r#"{"object":"PRODUCTION.employees","primary_index":{"type":"UPI","type_label":"Unique Primary Index (UPI)","columns":["employee_id"],"no_pi":false},"secondary_indexes":[]}"#;
        let parsed: serde_json::Value = serde_json::from_str(json).unwrap();
        assert!(parsed["object"].is_string(), "JSON must have 'object' string key");
    }

    // TC-047-006-L: JSON primary_index has required subkeys (AC-3)
    #[test]
    fn show_indexes_json_primary_index_has_subkeys() {
        let json = r#"{"object":"PRODUCTION.employees","primary_index":{"type":"UPI","type_label":"Unique Primary Index (UPI)","columns":["employee_id"],"no_pi":false},"secondary_indexes":[]}"#;
        let parsed: serde_json::Value = serde_json::from_str(json).unwrap();
        let primary = &parsed["primary_index"];
        assert!(primary["type"].is_string(), "primary_index must have 'type' key");
        assert!(primary["type_label"].is_string(), "primary_index must have 'type_label' key");
        assert!(primary["columns"].is_array(), "primary_index must have 'columns' array");
        assert!(primary["no_pi"].is_boolean(), "primary_index must have 'no_pi' boolean");
    }

    // TC-047-006-M: JSON secondary_indexes is an array (AC-3)
    #[test]
    fn show_indexes_json_secondary_indexes_is_array() {
        let json = r#"{"object":"PRODUCTION.employees","primary_index":{"type":"UPI","type_label":"Unique Primary Index (UPI)","columns":["employee_id"],"no_pi":false},"secondary_indexes":[{"index_no":1,"type":"NUSI","type_label":"Non-Unique Secondary Index (NUSI)","columns":["department_id"]}]}"#;
        let parsed: serde_json::Value = serde_json::from_str(json).unwrap();
        assert!(
            parsed["secondary_indexes"].is_array(),
            "secondary_indexes must be an array"
        );
    }

    // TC-047-006-N: JSON NoPI table has no_pi=true, type=null, columns=[] (AC-3)
    #[test]
    fn show_indexes_json_nopi_table_structure() {
        let json = r#"{"object":"PRODUCTION.fact_sales","primary_index":{"type":null,"type_label":"No Primary Index (NoPI)","columns":[],"no_pi":true},"secondary_indexes":[]}"#;
        let parsed: serde_json::Value = serde_json::from_str(json).unwrap();
        let primary = &parsed["primary_index"];
        assert!(
            primary["no_pi"].as_bool() == Some(true),
            "NoPI table must have no_pi=true — got: {:?}",
            primary["no_pi"]
        );
        assert!(
            primary["type"].is_null(),
            "NoPI table must have type=null — got: {:?}",
            primary["type"]
        );
        assert!(
            primary["columns"].as_array().map(|a| a.is_empty()).unwrap_or(false),
            "NoPI table must have empty columns array"
        );
    }

    // TC-047-006-O: CSV header row is correct (AC-3)
    #[test]
    fn show_indexes_csv_header_is_correct() {
        let header = "kind,index_no,type,columns";
        assert_eq!(header, "kind,index_no,type,columns", "CSV header must be 'kind,index_no,type,columns'");
    }
}

// TC-047-006-P: tq show-indexes shows Primary and Secondary sections
#[test]
#[ignore]
fn show_indexes_shows_primary_and_secondary_sections() {
    let output = Command::new(tq_binary())
        .args(["-l", &logon(), "show-indexes", "dbc.tables"])
        .output()
        .expect("Failed to run tq show-indexes");

    let stdout = String::from_utf8_lossy(&output.stdout);
    assert!(output.status.success(), "tq show-indexes must exit 0 for valid table");
    assert!(
        stdout.contains("Primary Index"),
        "Output must contain 'Primary Index' section — got: {}",
        &stdout[..300.min(stdout.len())]
    );
    assert!(
        stdout.contains("Secondary Indexes") || stdout.contains("No secondary"),
        "Output must contain 'Secondary Indexes' section or 'No secondary indexes' message"
    );
}

// TC-047-006-Q: tq show-indexes shows UPI/NUPI/USI/NUSI labels
#[test]
#[ignore]
fn show_indexes_shows_upi_nupi_usi_nusi_labels() {
    let output = Command::new(tq_binary())
        .args(["-l", &logon(), "show-indexes", "dbc.tables"])
        .output()

// TC-047-006-R: tq show-indexes --format json returns structured object
#[test]
#[ignore]
fn show_indexes_json_format_returns_structured_object() {
    let output = Command::new(tq_binary())
        .args(["-l", &logon(), "show-indexes", "--format", "json", "dbc.tables"])
        .output()
        .expect("Failed to run tq show-indexes --format json");

    let stdout = String::from_utf8_lossy(&output.stdout);
    assert!(output.status.success());

    let json: serde_json::Value = serde_json::from_str(&stdout)
        .unwrap_or_else(|e| panic!("Output must be valid JSON: {}\nGot: {}", e, &stdout[..200.min(stdout.len())]));

    assert!(json["object"].is_string(), "JSON must have 'object' key");
    assert!(
        !json["primary_index"].is_null(),
        "JSON must have 'primary_index' key"
    );
    assert!(
        json["secondary_indexes"].is_array(),
        "JSON must have 'secondary_indexes' array"
    );
}

// TC-047-006-S: tq show-indexes nonexistent exits 1 with Error: message
#[test]
#[ignore]
fn show_indexes_nonexistent_object_exits_1_with_error() {

        .output()
        .expect("Failed to run tq show-indexes on nonexistent object");

    let stdout = String::from_utf8_lossy(&output.stdout);
    let stderr = String::from_utf8_lossy(&output.stderr);

    assert_eq!(
        output.status.code(),
        Some(1),
        "Must exit with code 1 for not-found object"
    );
    assert!(
        stdout.contains("Error:") || stderr.contains("Error:"),
        "Not-found message must contain 'Error:'"
    );
}
