## Purpose

Verify that the `tq profiles` command correctly displays profiles from both user config and project config with clear source indicators.

## Acceptance Criteria Coverage

- **AC-4**: `tq profiles` command shows both user and project profiles - Command output
- **AC-8**: Test coverage: unit tests for config loading, integration tests for profile resolution - Integration test component

## Scope

This test validates:
- User-only profiles listed under "From user config"
- Project-only profiles listed under "From project config"
- Merged profiles listed under "From both (merged)" with field source indicators
- Clear source attribution ([user] and [project] tags)
- All profile fields displayed correctly
- Output format is user-friendly

## Prerequisites

- Rust test framework available
- tempfile crate available
- `tq` binary compiled
- Config loading implementation complete

## Test Procedure

### Test Implementation:

```rust
#[cfg(test)]
mod tests {
    use std::fs;
    use std::process::Command;
    use tempfile::TempDir;

    #[test]
    fn test_profiles_command_user_config_only() {
        // Create temp directory with user config only
        let temp_dir = TempDir::new().unwrap();
        let user_config_dir = temp_dir.path().join(".tq");
        fs::create_dir(&user_config_dir).unwrap();

        let user_config_path = user_config_dir.join("config.toml");
        let user_config_content = r#"
            [profiles.dev]
            host = "localhost"
            database = "mydev"
            user = "alice"

            [profiles.personal]
            host = "my-home-server.local"
            database = "sandbox"
        "#;
        fs::write(&user_config_path, user_config_content).unwrap();

        // Run tq profiles with custom home directory
        let output = Command::new("cargo")
            .args(&["run", "--", "profiles"])
            .env("HOME", temp_dir.path())
            .output()
            .expect("Failed to execute tq profiles");

        let stdout = String::from_utf8_lossy(&output.stdout);

        // Verify output
        assert!(output.status.success(), "Command should succeed");
        assert!(stdout.contains("Available profiles"), "Should show header");
        assert!(
            stdout.contains("From user config"),
            "Should show user config section"
        );
        assert!(stdout.contains("dev"), "Should list dev profile");
        assert!(stdout.contains("personal"), "Should list personal profile");
        assert!(stdout.contains("localhost"), "Should show dev host");
        assert!(
            stdout.contains("my-home-server.local"),
            "Should show personal host"
        );
    }

    #[test]
    fn test_profiles_command_project_config_only() {
        // Create temp directory with project config only
        let temp_dir = TempDir::new().unwrap();
        let project_config_path = temp_dir.path().join(".tq.toml");
        let project_config_content = r#"
            [profiles.dev]
            host = "dev.company.com"
            database = "shared_dev"

            [profiles.staging]
            host = "staging.company.com"
            database = "staging"
        "#;
        fs::write(&project_config_path, project_config_content).unwrap();

        // Run tq profiles from project directory (no user config)
        let output = Command::new("cargo")
            .args(&["run", "--", "profiles"])
            .current_dir(temp_dir.path())
            .env("HOME", "/nonexistent") // Ensure no user config loaded
            .output()
            .expect("Failed to execute tq profiles");

        let stdout = String::from_utf8_lossy(&output.stdout);

        // Verify output
        assert!(output.status.success(), "Command should succeed");
        assert!(
            stdout.contains("From project config"),
            "Should show project config section"
        );
        assert!(stdout.contains("dev"), "Should list dev profile");
        assert!(stdout.contains("staging"), "Should list staging profile");
        assert!(
            stdout.contains("dev.company.com"),
            "Should show dev host"
        );
        assert!(
            stdout.contains("staging.company.com"),
            "Should show staging host"
        );
    }

    #[test]
    fn test_profiles_command_both_configs() {
        // Create temp directory with both user and project configs
        let temp_dir = TempDir::new().unwrap();

        // User config
        let user_config_dir = temp_dir.path().join(".tq");
        fs::create_dir(&user_config_dir).unwrap();
        let user_config_path = user_config_dir.join("config.toml");
        let user_config_content = r#"
            [profiles.dev]
            host = "localhost"
            user = "alice"
            password_file = "~/.tq/passwords/dev"

            [profiles.personal]
            host = "my-home-server.local"
            database = "sandbox"
        "#;
        fs::write(&user_config_path, user_config_content).unwrap();

        // Project config
        let project_config_path = temp_dir.path().join(".tq.toml");
        let project_config_content = r#"
            [profiles.dev]
            host = "dev.company.com"
            database = "shared_dev"

            [profiles.staging]
            host = "staging.company.com"
            database = "staging"
        "#;
        fs::write(&project_config_path, project_config_content).unwrap();

        // Run tq profiles
        let output = Command::new("cargo")
            .args(&["run", "--", "profiles"])
            .current_dir(temp_dir.path())
            .env("HOME", temp_dir.path())
            .output()
            .expect("Failed to execute tq profiles");

        let stdout = String::from_utf8_lossy(&output.stdout);

        // Verify output structure
        assert!(output.status.success(), "Command should succeed");
        assert!(stdout.contains("Available profiles"), "Should show header");

        // User-only section
        assert!(
            stdout.contains("From user config"),
            "Should have user config section"
        );
        assert!(stdout.contains("personal"), "Should list personal profile");

        // Project-only section
        assert!(
            stdout.contains("From project config"),
            "Should have project config section"
        );
        assert!(stdout.contains("staging"), "Should list staging profile");

        // Merged section (dev profile in both configs)
        assert!(
            stdout.contains("From both (merged)") || stdout.contains("dev"),
            "Should show merged dev profile"
        );
        assert!(
            stdout.contains("[project]") || stdout.contains("[user]"),
            "Should show source indicators for merged profile"
        );

        // Verify profile details
        assert!(
            stdout.contains("dev.company.com"),
            "Should show project host for dev"
        );
        assert!(
            stdout.contains("shared_dev"),
            "Should show project database for dev"
        );
        assert!(
            stdout.contains("alice"),
            "Should show user username for dev"
        );
    }

    #[test]
    fn test_profiles_command_with_profile_name_conflict() {
        // Create configs with same profile name (dev)
        let temp_dir = TempDir::new().unwrap();

        // User config: dev profile
        let user_config_dir = temp_dir.path().join(".tq");
        fs::create_dir(&user_config_dir).unwrap();
        let user_config_path = user_config_dir.join("config.toml");
        let user_config_content = r#"
            [profiles.dev]
            host = "localhost"
            database = "my_local_dev"
            user = "alice"
        "#;
        fs::write(&user_config_path, user_config_content).unwrap();

        // Project config: dev profile (different values)
        let project_config_path = temp_dir.path().join(".tq.toml");
        let project_config_content = r#"
            [profiles.dev]
            host = "dev.company.com"
            database = "shared_dev"
            logmech = "LDAP"
        "#;
        fs::write(&project_config_path, project_config_content).unwrap();

        // Run tq profiles
        let output = Command::new("cargo")
            .args(&["run", "--", "profiles"])
            .current_dir(temp_dir.path())
            .env("HOME", temp_dir.path())
            .output()
            .expect("Failed to execute tq profiles");

        let stdout = String::from_utf8_lossy(&output.stdout);

        // Verify merged profile shows project values take precedence
        assert!(output.status.success());
        assert!(
            stdout.contains("dev.company.com"),
            "Should show project host (not localhost)"
        );
        assert!(
            stdout.contains("shared_dev"),
            "Should show project database (not my_local_dev)"
        );
        assert!(
            stdout.contains("alice"),
            "Should show user username (inherited)"
        );

        // Should indicate field sources
        if stdout.contains("[project]") && stdout.contains("[user]") {
            // Bonus points for clear source attribution
            assert!(true);
        }
    }

    #[test]
    fn test_profiles_command_no_config() {
        // Run tq profiles with no configs
        let temp_dir = TempDir::new().unwrap();

        let output = Command::new("cargo")
            .args(&["run", "--", "profiles"])
            .current_dir(temp_dir.path())
            .env("HOME", "/nonexistent")
            .output()
            .expect("Failed to execute tq profiles");

        let stdout = String::from_utf8_lossy(&output.stdout);

        // Should succeed with message about no profiles
        assert!(output.status.success());
        assert!(
            stdout.contains("No profiles") || stdout.contains("no profiles configured"),
            "Should indicate no profiles found"
        );
    }
}
```
