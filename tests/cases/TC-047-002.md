## Purpose

Verify that:
1. All four utility functions exist exactly once — in `src/commands/format_helpers.rs`.
2. No duplicate definitions remain in `describe.rs`, `list.rs`, `show_indexes.rs`, or `inspect.rs`.
3. `truncate_str` uses `char_indices()` for UTF-8 safety (the previous byte-slice implementation panics on multi-byte characters).
4. All functions behave identically to their previous per-module implementations (no regression).

The UTF-8 boundary test is the critical regression guard: `truncate_str("éàü_suffix", 4)` must not panic and must return valid UTF-8.

## Acceptance Criteria Coverage

- **AC-1**: `json_escape()` in format_helpers.rs only
- **AC-2**: `csv_escape()` in format_helpers.rs only
- **AC-3**: `parse_table_name()` in format_helpers.rs (or identifiers.rs) once
- **AC-4**: `truncate_str()` with UTF-8 `char_indices()` handling
- **AC-5**: Zero duplication
- **AC-6**: All existing tests pass after extraction

## Part A — Unit Tests (No DB Required)

### Prerequisites

- `src/commands/format_helpers.rs` exists and is declared in `src/commands/mod.rs`
- All four functions are `pub fn` in `format_helpers.rs`
- `cargo test --lib` compiles successfully

### Test Implementation

Create the following test module in `src/commands/format_helpers.rs`:

```rust
#[cfg(test)]
mod tests {
    use super::*;

    // =========================================================================
    // json_escape tests (AC-1)
    // =========================================================================

    // TC-047-002-A: plain string — no escaping needed
    #[test]
    fn json_escape_plain_string_unchanged() {
        assert_eq!(json_escape("hello"), "hello");
    }

    // TC-047-002-B: empty string
    #[test]
    fn json_escape_empty_string() {
        assert_eq!(json_escape(""), "");
    }

    // TC-047-002-C: double quote escaped
    #[test]
    fn json_escape_double_quote() {
        assert_eq!(json_escape("he\"llo"), "he\\\"llo");
    }

    // TC-047-002-D: newline escaped
    #[test]
    fn json_escape_newline() {
        assert_eq!(json_escape("line\nnew"), "line\\nnew");
    }

    // TC-047-002-E: tab escaped
    #[test]
    fn json_escape_tab() {
        assert_eq!(json_escape("tab\there"), "tab\\there");
    }

    // TC-047-002-F: backslash escaped (must be doubled)
    #[test]
    fn json_escape_backslash() {
        assert_eq!(json_escape("back\\slash"), "back\\\\slash");
    }

    // =========================================================================
    // csv_escape tests (AC-2)
    // =========================================================================

    // TC-047-002-G: plain string — no quoting needed
    #[test]
    fn csv_escape_plain_string_unchanged() {
        assert_eq!(csv_escape("hello"), "hello");
    }

    // TC-047-002-H: string with comma — wrapped in quotes
    #[test]
    fn csv_escape_comma_wraps_in_quotes() {
        assert_eq!(csv_escape("hello,world"), "\"hello,world\"");
    }

    // TC-047-002-I: string with embedded double quote — quote doubled and wrapped
    #[test]
    fn csv_escape_embedded_quote_doubled() {
        assert_eq!(csv_escape("say \"hi\""), "\"say \"\"hi\"\"\"");
    }

    // TC-047-002-J: string with newline — wrapped in quotes
    #[test]
    fn csv_escape_newline_wraps_in_quotes() {
        let result = csv_escape("line\nnew");
        assert!(result.starts_with('"'), "Newline value must be CSV-quoted");
        assert!(result.ends_with('"'), "Newline value must be CSV-quoted");
    }

    // =========================================================================
    // parse_table_name tests (AC-3)
    // =========================================================================

    // TC-047-002-K: unqualified name — no dot
    #[test]
    fn parse_table_name_unqualified() {
        let (db, obj) = parse_table_name("mytable");
        assert!(db.is_none(), "Unqualified name must have no database part");
        assert_eq!(obj, "mytable");
    }

    // TC-047-002-L: qualified name — first dot is separator
    #[test]
    fn parse_table_name_qualified() {
        let (db, obj) = parse_table_name("mydb.mytable");
        assert_eq!(db, Some("mydb"));
        assert_eq!(obj, "mytable");
    }

    // TC-047-002-M: multiple dots — first dot wins
    #[test]
    fn parse_table_name_multiple_dots_first_wins() {
        let (db, obj) = parse_table_name("a.b.c");
        assert_eq!(db, Some("a"));
        assert_eq!(obj, "b.c");
    }

    // =========================================================================
    // truncate_str tests (AC-4)
    // =========================================================================

    // TC-047-002-N: string shorter than max_len — returned as-is
    #[test]
    fn truncate_str_short_string_unchanged() {
        assert_eq!(truncate_str("short", 10), "short");
    }

    // TC-047-002-O: string exactly at max_len — returned as-is (no ellipsis)
    #[test]
    fn truncate_str_exact_fit_unchanged() {
        assert_eq!(truncate_str("exactly10c", 10), "exactly10c");
    }

    // TC-047-002-P: string longer than max_len — truncated with "..."
    #[test]
    fn truncate_str_long_ascii_truncated() {
        assert_eq!(truncate_str("this is a long string", 10), "this is...");
    }

    // TC-047-002-Q: UTF-8 safety — 2-byte chars (é = U+00E9, 2 bytes in UTF-8)
    // max_len=5 means result should be 5 *characters* with "..." suffix (so 2 chars + "...")
    // The old byte-slice implementation would panic if a 2-byte char straddles the cut point.
    #[test]
    fn truncate_str_utf8_two_byte_chars_no_panic() {
        // "éàüöñ" = 5 chars, each 2 bytes = 10 bytes total
        // Truncating to 4 chars: "é..." (1 char + "...")
        let s = "éàüöñ";
        let result = truncate_str(s, 4);
        // Must be valid UTF-8 (no panic is the primary assertion — test framework catches panics)
        assert!(std::str::from_utf8(result.as_bytes()).is_ok(), "Result must be valid UTF-8");
        assert!(result.len() <= s.len(), "Truncated result must not be longer than input");
        // Must end with "..." (we truncated)
        assert!(result.ends_with("..."), "Long string must end with '...' after truncation: got '{}'", result);
    }

    // TC-047-002-R: UTF-8 safety — 3-byte CJK characters
    // CJK chars are 3 bytes each; old byte-slice truncation would panic mid-codepoint
    #[test]
    fn truncate_str_utf8_three_byte_cjk_no_panic() {
        // "中文字符串测试" = 7 CJK chars, each 3 bytes = 21 bytes
        let s = "中文字符串测试";
        let result = truncate_str(s, 5);
        assert!(std::str::from_utf8(result.as_bytes()).is_ok(), "Result must be valid UTF-8");
        assert!(result.ends_with("..."), "Long CJK string must end with '...' after truncation: got '{}'", result);
    }

    // TC-047-002-S: max_len = 3 — result is "..." (full ellipsis)
    #[test]
    fn truncate_str_max_len_equals_three() {
        let result = truncate_str("abcde", 3);
        assert_eq!(result.len(), 3, "Result must be exactly 3 bytes");
        // When max_len <= 3, the entire result is dots
        assert!(result.chars().all(|c| c == '.'), "All characters must be dots when max_len=3");
    }

    // TC-047-002-T: max_len = 2 — result is ".."
    #[test]
    fn truncate_str_max_len_less_than_three() {
        let result = truncate_str("abcde", 2);
        assert_eq!(result, "..", "max_len=2 must return '..'");
    }
}
```
