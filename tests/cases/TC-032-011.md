## Purpose

Verify that content-based column width calculation integrates correctly with the complete table formatting pipeline, producing well-formed table output.

## Acceptance Criteria Coverage

- **AC-1**: Column width calculated from actual cell content, not schema type
- **AC-5**: Existing table formatting tests still pass

## Scope

This test validates:
- End-to-end table formatting with content-based widths
- Integration with existing table renderer
- Table structure correctness (borders, alignment, spacing)
- No regressions in table formatting

## Prerequisites

- Rust test framework available
- Access to table formatting module in `src/format/table.rs`
- Mock `QueryResult` creation

#[test]
fn test_content_based_width_end_to_end() {
    // Setup: Create QueryResult with VARCHAR(64) columns containing short content
    let columns = vec![
        ColumnMetadata {
            name: "DatabaseName".to_string(),
            type_info: "VARCHAR(64)".to_string(),
            nullable: false,
        },
        ColumnMetadata {
            name: "OwnerName".to_string(),
            type_info: "VARCHAR(64)".to_string(),
            nullable: false,
        },
        ColumnMetadata {
            name: "AccountName".to_string(),
            type_info: "VARCHAR(64)".to_string(),
            nullable: false,
        },
    ];
    let rows = vec![
        vec![
            Value::String("SystemDB".to_string()),    // 8 chars
            Value::String("DBC".to_string()),         // 3 chars
            Value::String("$SYSTEM".to_string()),     // 7 chars
        ],
        vec![
            Value::String("TempDB".to_string()),      // 6 chars
            Value::String("DBC".to_string()),         // 3 chars
            Value::String("$SYSTEM".to_string()),     // 7 chars
        ],
    ];
    let result = QueryResult::new(columns, rows, Duration::from_millis(45));

    // Action: Format as table
    let formatter = TableFormatter::new();
    let output = formatter.format(&result, None); // No terminal width limit

    // Assert: Table structure is correct
    assert!(output.contains("┌─"), "Should have top border");
    assert!(output.contains("│ DatabaseName"), "Should have DatabaseName header");
    assert!(output.contains("│ OwnerName"), "Should have OwnerName header");
    assert!(output.contains("│ AccountName"), "Should have AccountName header");
    assert!(output.contains("├─"), "Should have header separator");
    assert!(output.contains("│ SystemDB"), "Should have SystemDB row");
    assert!(output.contains("│ TempDB"), "Should have TempDB row");
    assert!(output.contains("└─"), "Should have bottom border");

    // Assert: Columns are not padded to 64 characters (schema width)
    // Count characters between first two column separators
    let lines: Vec<&str> = output.lines().collect();
    let data_line = lines.iter().find(|l| l.contains("SystemDB")).expect("Data line");

    // Column should NOT be 64 chars wide (would waste space)
    // Instead, should be roughly ~12 chars (header "DatabaseName" = 12)
    assert!(
        !data_line.contains(&" ".repeat(50)),
        "Column should not have excessive padding (schema width)"
    );
}

#[test]
fn test_table_formatting_with_mixed_types() {
    // Setup: Multiple column types
    let columns = vec![
        ColumnMetadata::new("ID", "INTEGER"),
        ColumnMetadata::new("Name", "VARCHAR(50)"),
        ColumnMetadata::new("Active", "BOOLEAN"),
        ColumnMetadata::new("Amount", "DECIMAL(10,2)"),
    ];
    let rows = vec![
        vec![
            Value::Integer(1),
            Value::String("Alice".to_string()),
            Value::Boolean(true),
            Value::Float(123.45),
        ],
        vec![
            Value::Integer(42),
            Value::String("Bob".to_string()),
            Value::Boolean(false),
            Value::Float(9.99),
        ],
    ];
    let result = QueryResult::new(columns, rows, Duration::from_millis(20));

    let formatter = TableFormatter::new();
    let output = formatter.format(&result, None);

    // Assert: All columns present
    assert!(output.contains("│ ID"), "Should have ID column");
    assert!(output.contains("│ Name"), "Should have Name column");
    assert!(output.contains("│ Active"), "Should have Active column");
    assert!(output.contains("│ Amount"), "Should have Amount column");

    // Assert: Table structure intact
    assert!(output.contains("┌─"), "Top border");
    assert!(output.contains("├─"), "Header separator");
    assert!(output.contains("└─"), "Bottom border");

    // Assert: No formatting corruption
    assert_eq!(output.matches("┌─").count(), 1, "Should have one top border");
    assert_eq!(output.matches("└─").count(), 1, "Should have one bottom border");
}

#[test]
fn test_existing_table_formatter_still_works() {
    // Regression test: Verify existing table formatting behavior preserved
    let columns = vec![
        ColumnMetadata::new("id", "INTEGER"),
        ColumnMetadata::new("name", "VARCHAR(20)"),
    ];
    let rows = vec![
        vec![Value::Integer(1), Value::String("Alice".to_string())],
        vec![Value::Integer(2), Value::String("Bob".to_string())],
    ];
    let result = QueryResult::new(columns, rows, Duration::from_millis(10));

    let formatter = TableFormatter::new();
    let output = formatter.format(&result, None);

    // Assert: Basic formatting still works (regression check)
    assert!(output.contains("┌─"), "Top border present");
    assert!(output.contains("│"), "Column separators present");
    assert!(output.contains("├─"), "Header separator present");
    assert!(output.contains("└─"), "Bottom border present");
    assert!(output.contains("Alice"), "Data row 1 present");
    assert!(output.contains("Bob"), "Data row 2 present");
}

- Any integration test fails
- Table structure corrupted (missing borders, misaligned)
- Columns still padded to schema width (regression)
