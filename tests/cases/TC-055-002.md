## Purpose

Verify that:
1. The `ColumnSearchResult` struct is correctly defined with all required fields.
2. All four rendering functions produce correct output with populated results.
3. The JSON `nullable` field is a JSON boolean (`true`/`false`), not a string.
4. All four rendering functions handle empty results gracefully.
5. Live database search by column keyword returns results.
6. `--database` flag correctly scopes results.
7. No-results case exits cleanly.

// TC-055-002-B3: JSON format with results uses standard envelope and boolean nullable
#[test]
fn render_search_columns_json_envelope_and_nullable_boolean() {
    let results = make_column_search_results();
    let mut buf: Vec<u8> = Vec::new();
    render_search_columns_json(&results, &mut buf).unwrap();
    let output = String::from_utf8(buf).unwrap();

    let json: serde_json::Value = serde_json::from_str(&output)
        .unwrap_or_else(|e| panic!("Output must be valid JSON: {}\nOutput: {}", e, output));

    // Standard envelope
    assert_eq!(json["ok"], true, "Envelope must have ok: true");
    assert_eq!(json["row_count"], 2, "Envelope must have row_count: 2");
    assert!(json["data"].is_array(), "Envelope must have data array");

    let data = json["data"].as_array().unwrap();
    assert_eq!(data.len(), 2, "data must have 2 entries");

    // CRITICAL: nullable must be a JSON boolean, not a string
    for (i, entry) in data.iter().enumerate() {
        let nullable_val = &entry["nullable"];
        assert!(
            nullable_val.is_boolean(),
            "Entry {}: nullable must be a JSON boolean (not a string), got: {:?}",
            i, nullable_val
        );
    }

    // First entry: nullable: true (boolean)
    assert_eq!(data[0]["nullable"], true,
        "First result nullable must be boolean true: {:?}", data[0]);

    // Second entry: nullable: false (boolean)
    assert_eq!(data[1]["nullable"], false,
        "Second result nullable must be boolean false: {:?}", data[1]);

    // Must not contain string "YES" or "NO" for nullable
    assert!(!output.contains("\"YES\""), "nullable must not be string 'YES': {}", output);
    assert!(!output.contains("\"NO\""), "nullable must not be string 'NO': {}", output);
}

/// TC-055-002-E1: tq search columns --format json produces valid JSON envelope with boolean nullable
///
/// Run with: cargo test test_search_columns_json_nullable_boolean -- --ignored
#[test]
#[ignore] // Requires live database
fn test_search_columns_json_nullable_boolean() {
    dotenvy::dotenv().ok();
    let logon = std::env::var("TQ_LOGON").expect("TQ_LOGON must be set");

    let output = assert_cmd::Command::cargo_bin("tq")
        .unwrap()
        .args(["-l", &logon, "search", "columns", "name", "--format", "json"])
        .output()
        .expect("Failed to run tq search columns --format json");

    assert!(output.status.success(), "Exit code must be 0");
    let stdout = String::from_utf8_lossy(&output.stdout);

    let json: serde_json::Value = serde_json::from_str(&stdout)
        .unwrap_or_else(|e| panic!("Output must be valid JSON: {}\nOutput: {}", e, stdout));

    // Standard envelope
    assert_eq!(json["ok"], true, "Envelope must have ok: true");
    assert!(json["row_count"].is_number(), "Envelope must have numeric row_count");
    assert!(json["data"].is_array(), "Envelope must have data array");

    // For any results returned, nullable must be a JSON boolean
    if let Some(data) = json["data"].as_array() {
        if !data.is_empty() {
            for (i, entry) in data.iter().enumerate() {
                if let Some(nullable_val) = entry.get("nullable") {
                    assert!(
                        nullable_val.is_boolean(),
                        "Entry {}: nullable must be a JSON boolean, got: {:?}",
                        i, nullable_val
                    );
                }
            }
        }
    }
}

## Pass/Fail Criteria

**PASS if:**
- All 8 unit tests pass without a database
- `nullable` field in JSON output is a JSON boolean literal (not a string "YES"/"NO")
- All 4 integration tests pass with a valid `TQ_LOGON`

**FAIL if:**
- Unit test finds `nullable` is not a boolean type — API contract violation
- CSV test finds unquoted `DECIMAL(15,2),` — comma escaping failure
- Integration test finds non-zero exit code for valid keyword
- Integration test finds `nullable` as string `"YES"` or `"NO"` in live output

**BLOCKED if:**
- `TQ_LOGON` not set — integration tests C-G cannot run; report BLOCKED for AC-1, AC-2, AC-5 end-to-end
- `src/commands/search.rs` does not yet exist — report BLOCKED
