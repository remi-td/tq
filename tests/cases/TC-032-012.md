## Purpose

Verify that content-based width calculation works correctly with terminal width truncation (column hiding when terminal is narrow).

## Acceptance Criteria Coverage

- **AC-1**: Column width calculated from actual cell content
- **AC-4**: More columns visible with content-based widths (implicit validation)

## Scope

This test validates:
- Content-based widths with terminal width limit

- Truncation indicator when columns hidden

#[test]
fn test_content_based_width_with_terminal_limit() {
    // Setup: 5 columns with short content in narrow terminal
    let columns = vec![
        ColumnMetadata::new("ID", "INTEGER"),
        ColumnMetadata::new("Name", "VARCHAR(50)"),       // Schema: 50, content: 7
        ColumnMetadata::new("Status", "VARCHAR(50)"),     // Schema: 50, content: 6
        ColumnMetadata::new("Code", "VARCHAR(20)"),       // Schema: 20, content: 4
        ColumnMetadata::new("Flag", "BOOLEAN"),
    ];
    let rows = vec![
        vec![
            Value::Integer(1),
            Value::String("Alice".to_string()),    // 5 chars
            Value::String("active".to_string()),   // 6 chars
            Value::String("A001".to_string()),     // 4 chars
            Value::Boolean(true),
        ],
        vec![
            Value::Integer(42),
            Value::String("Charlie".to_string()),  // 7 chars (max)
            Value::String("idle".to_string()),     // 4 chars
            Value::String("B022".to_string()),     // 4 chars
            Value::Boolean(false),
        ],
    ];
    let result = QueryResult::new(columns, rows, Duration::from_millis(10));

    // Action: Format with 80-char terminal width
    let formatter = TableFormatter::new();
    let output = formatter.format(&result, Some(80));

    // Assert: Multiple columns fit due to content-based widths
    // Approximate widths: ID(2) + Name(7) + Status(6) + Code(4) + Flag(4) = ~23 chars of data
    // Plus borders and spacing: ~35 chars total
    // Should fit at least 4 columns in 80-char terminal

    let visible_columns = count_visible_columns(&output);
    assert!(
        visible_columns >= 4,
        "At least 4 columns should be visible with content-based widths (found: {})",
        visible_columns
    );

    // If columns are hidden, truncation indicator should appear
    if visible_columns < 5 {
        assert!(
            output.contains("(+") || output.contains("cols"),
            "Should show truncation indicator when columns hidden"
        );
    }
}

#[test]
fn test_more_columns_visible_than_schema_based() {
    // Setup: Table where schema-based would show 2 cols, content-based shows 5+
    let columns = vec![
        ColumnMetadata::new("Col1", "VARCHAR(60)"),  // Schema: 60, content: 5
        ColumnMetadata::new("Col2", "VARCHAR(60)"),  // Schema: 60, content: 6
        ColumnMetadata::new("Col3", "VARCHAR(60)"),  // Schema: 60, content: 4
        ColumnMetadata::new("Col4", "VARCHAR(60)"),  // Schema: 60, content: 7
        ColumnMetadata::new("Col5", "VARCHAR(60)"),  // Schema: 60, content: 3
        ColumnMetadata::new("Col6", "VARCHAR(60)"),  // Schema: 60, content: 5
    ];
    let rows = vec![
        vec![
            Value::String("Data1".to_string()),  // 5 chars
            Value::String("Value2".to_string()), // 6 chars
            Value::String("Item".to_string()),   // 4 chars
            Value::String("Content".to_string()), // 7 chars
            Value::String("Val".to_string()),    // 3 chars
            Value::String("Field".to_string()),  // 5 chars
        ],
    ];
    let result = QueryResult::new(columns, rows, Duration::from_millis(10));

    // Action: Format with 120-char terminal (typical wide terminal)
    let formatter = TableFormatter::new();
    let output = formatter.format(&result, Some(120));

    // Assert: With content-based widths, should see 5+ columns
    // Schema-based: Each column 60 chars → only 1-2 cols fit
    // Content-based: Each column ~5-7 chars → 5+ cols fit
    let visible_columns = count_visible_columns(&output);
    assert!(
        visible_columns >= 5,
        "Content-based widths should show 5+ columns (found: {})",
        visible_columns
    );
}

#[test]
fn test_batch_mode_shows_all_columns() {
    // Setup: Many columns
    let columns = vec![
        ColumnMetadata::new("C1", "VARCHAR(50)"),
        ColumnMetadata::new("C2", "VARCHAR(50)"),
        ColumnMetadata::new("C3", "VARCHAR(50)"),
        ColumnMetadata::new("C4", "VARCHAR(50)"),
        ColumnMetadata::new("C5", "VARCHAR(50)"),
    ];
    let rows = vec![
        vec![
            Value::String("Val1".to_string()),
            Value::String("Val2".to_string()),
            Value::String("Val3".to_string()),
            Value::String("Val4".to_string()),
            Value::String("Val5".to_string()),
        ],
    ];
    let result = QueryResult::new(columns, rows, Duration::from_millis(10));

    // Action: Format with NO terminal width limit (batch mode)
    let formatter = TableFormatter::new();
    let output = formatter.format(&result, None);

    // Assert: All 5 columns visible
    let visible_columns = count_visible_columns(&output);
    assert_eq!(visible_columns, 5, "Batch mode should show all columns");

    // No truncation indicator
    assert!(
        !output.contains("(+") && !output.contains("hidden"),
        "Batch mode should not show truncation indicator"
    );
}

// Helper function
fn count_visible_columns(output: &str) -> usize {
    // Count column headers in output
    // Look for lines containing column names before "├─" separator
    let lines: Vec<&str> = output.lines().collect();
    let header_line = lines.iter()
        .find(|l| l.contains("│") && !l.contains("├") && !l.contains("└") && !l.contains("┌"))
        .expect("Header line");

    // Count column separators (│) - subtract 2 for left and right borders
    header_line.matches("│").count().saturating_sub(2)
}
